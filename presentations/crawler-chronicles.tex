\section{The premises}
\begin{frame}{One server, several ways to be visited}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \CLabel{The premises}
      A modest rented virtual server hosts both the website and
      \texttt{git.levineuwirth.org}.\par\medskip
      \textbf{Forgejo} is the software that serves the code repositories:
      files, history, and downloadable archives.\par\medskip
      \CAside{I am a researcher. The server is not a hyperscaler.}
    \end{column}
    \begin{column}{.46\textwidth}
      \CLabel{A small vocabulary}
      \textbf{Repository:} a project and its history.\par\smallskip
      \textbf{Commit:} a recorded version.\par\smallskip
      \textbf{Crawler:} software visiting pages automatically.\par\medskip
      \textbf{HTTP 200:} a successful response.\par\smallskip
      \textbf{HTTP 403:} a challenge or denial in these logs.
    \end{column}
  \end{columns}
  \note{A request is not a person, and HTTP 200 alone does not establish that a visitor was a bot. The early tables count repository-path responses across agents; the later test uses a much narrower cohort.}
\end{frame}

% 21. Preserve dependence rather than narrating thousands of independent bots.
\begin{frame}{Move the clock; leave the swarm alone}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \CLabel{The problem}
      Crawlers arrive in bursts. Thousands of correlated requests
      are not thousands of independent experiments.\par\medskip
      A busy half hour might have been busy even without a push.
    \end{column}
    \begin{column}{.46\textwidth}
      \CLabel{The comparison}
      Keep the request timeline fixed. Shift publication times
      around within their occupied UTC hours.\par\medskip
      Pushes sharing an hour move together.\par\medskip
      Repeat \textbf{10,000 times} and compare the actual alignment
      with these shifted alignments.
    \end{column}
  \end{columns}
  \medskip\CAside{An observational comparison, conditional on this within-hour timing model.}
  \note{Circular rotation preserves the within-hour spacing modulo sixty minutes. It assumes within-hour exchangeability; the schedule was not randomized. Avoid describing the p-value as the probability that bots are monitoring the server.}
\end{frame}
