% --- lab controls ---------------------------------------------------------
% \atlascontrol{<name>}{<treatment>}{<modern default>} sets the treatment
% unless <name> is in \atlasomit, a comma list build.sh --ablate passes on
% the command line. \atlasstamp records the controls at the foot of the plate
% so every variant names itself. Identical in every example, by design.
\ExplSyntaxOn
\providecommand{\atlasomit}{}
\clist_new:N \l_atlas_omit_clist
\clist_new:N \g_atlas_seen_clist
\clist_set:Ne \l_atlas_omit_clist { \atlasomit }
\NewDocumentCommand{\atlascontrol}{ m +m +m }
  {
    \clist_gput_right:Nn \g_atlas_seen_clist { #1 }
    \clist_if_in:NnTF \l_atlas_omit_clist { #1 } { #3 } { #2 }
  }
\NewDocumentCommand{\atlasstamp}{}
  {
    \clist_remove_duplicates:N \g_atlas_seen_clist
    Controls:~\clist_use:Nn \g_atlas_seen_clist { ,~ }.
    \clist_if_empty:NF \l_atlas_omit_clist
      { ~This~impression~sets~\clist_use:Nn \l_atlas_omit_clist { ,~ }~to~the~modern~default. }
  }
\ExplSyntaxOff

% Paper tint is per-example tuning for this A5 treatise page, not a shared
% token yet; the examples deliberately differ slightly (see README). The
% control swaps the token values, so everything downstream stays unchanged.
\atlascontrol{paper}{
  \definecolor{parchment}{HTML}{F5EDDC}
  \definecolor{ink}{HTML}{241F19}
}{
  \definecolor{parchment}{HTML}{FFFFFF}
  \definecolor{ink}{HTML}{000000}
}
% The second colour. Without it every rubricated device stays, in ink.
\atlascontrol{rubric}{
  \definecolor{rubric}{HTML}{9E2B25}
}{
  \colorlet{rubric}{ink}
}
\pagecolor{parchment}
\color{ink}
\geometry{a5paper,top=20mm,bottom=22mm,left=22mm,right=22mm}

\setlength{\parskip}{0pt}

% Pilcrow as a section marker inside running prose; the modern default is
% the paragraph indent alone.
\atlascontrol{pilcrow}{
  \newcommand{\rubricp}{\textcolor{rubric}{\P}\ }
}{
  \newcommand{\rubricp}{}
}

% Dropped three-line initial via lettrine, or an ordinary first line.
\atlascontrol{initial}{
  \newcommand{\incipit}[2]{\lettrine[lines=3,loversize=0.08,findent=0.35em,nindent=0em]{\textcolor{rubric}{#1}}{#2}}
}{
  \newcommand{\incipit}[2]{\noindent #1#2}
}

% Running head in spaced small capitals under a hair rule, or the article
% class's plain page style.
\atlascontrol{runhead}{
  \pagestyle{fancy}
  \fancyhf{}
  \fancyhead[C]{\small\textls[90]{\textcolor{rubric}{\textsc{Of the Loadstone}}}}
  \fancyfoot[C]{\small\textit{1}}
  \renewcommand{\headrulewidth}{0.35pt}
  \renewcommand{\footrulewidth}{0pt}
}{
  \pagestyle{plain}
}

\begin{document}

\begin{center}
  {\small\textls[130]{\textsc{A Short Treatise}}}\par
  \vspace{0.3em}
  {\Large\textit{Of the Loadstone}}\par
  \vspace{0.3em}
  {\footnotesize\textcolor{rubric}{Rubricated after the manner of the early presses}}
\end{center}

\vspace{0.8em}

\incipit{W}{hen} the stone is hung by a thread and left to
settle, it turns its one end toward the north. This virtue is not spent by
use, for after a hundred trials the direction is the same as at the first.

\rubricp The trial is best made in a closed room, the air being still, the
iron being kept at a distance. Note the bearing before any cause is
assigned; conjecture follows observation, not the reverse.

\rubricp A second stone, floated upon a chip of wood in still water, obeys
the same rule. The agreement of the thread and the water is the whole of
the proof that the effect lies in the stone and not in the suspension.

\vspace{1em}
\begin{center}
  \textcolor{rubric}{\rule{0.3\linewidth}{0.45pt}}
\end{center}

\footnotesize\noindent
One colour and three devices to carry it: a three-line initial via
\texttt{lettrine}, pilcrows marking breaths in the argument, and a running
head with a hair rule. Body ink and measure are unchanged from a plain page.
